1. 明文 payload 字段

加密前是一份 UTF-8 JSON。字段名必须与下表一致。timestamp / expireAt 是数字不是字符串。

字段类型必填说明
userIdString是A 侧用户唯一 ID
userNameString是姓名 / 登录名
nickNameString是界面展示名，可与 userName 相同
tenantIdString是租户 ID
tenantCodeString是租户编码
tenantNameString是单位全称
regionCodeString是行政区划代码，如 420000 / 420100 / 420500
regionNameString建议湖北省 / 武汉市 / 宜昌市
timestamplong是签发时间，毫秒 UTC
expireAtlong是timestamp + 300000（5 分钟）
nonceString是UUID v4，每次新值，一次性
remarkString否如 from-alert。不要放密码、Token、身份证
首期没有 role。不要把密钥写进前端。IV 每次随机，verifyData 用 Base64URL（无 padding），AAD = clientId 的 UTF-8 字节。

3. 联调参数

clientId    = CDSS-B-ZJJK-001
keyVersion  = 1
密钥(标准Base64, 32字节) = luueY71IMszzt4bL8PocbEvkk1hXVpkY8Rpt2RxggUw=
算法        = AES-256-GCM
IV          = 12 字节随机（每次新生成）
Tag         = 128 bit
AAD         = "CDSS-B-ZJJK-001".getBytes("UTF-8")
verifyData  = Base64URL( keyVersion(1B) || IV(12B) || ciphertext||tag )
跳转        = https://<B域名>/hbcdcagent/workbench/deepseek?clientId={clientId}&verifyData={verifyData}
生产密钥线下另交，禁止把联调密钥带上生产。verifyData 已是 URL 安全编码，无需再 encodeURIComponent。

4. 可粘贴 Java（整类复制到 A 后端）

JDK 8+，只用 javax.crypto。调用 SsoTicketIssuer.issue(...) 后把 clientId + verifyData 给前端跳转。若项目已有 Fastjson / Jackson，可用其序列化 Map，但字段名必须与第 1 节一致。

import javax.crypto.Cipher;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.nio.ByteBuffer;
import java.nio.charset.StandardCharsets;
import java.security.SecureRandom;
import java.util.Base64;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;

public final class SsoTicketIssuer {

    public static final String CLIENT_ID = "CDSS-B-ZJJK-001";
    public static final int KEY_VERSION = 1;
    public static final String KEY_BASE64 = "luueY71IMszzt4bL8PocbEvkk1hXVpkY8Rpt2RxggUw=";
    public static final long TTL_MS = 300_000L;

    private static final int IV_LEN = 12;
    private static final int TAG_BITS = 128;
    private static final SecureRandom RANDOM = new SecureRandom();

    private SsoTicketIssuer() {}

    public static final class Ticket {
        public final String clientId;
        public final String verifyData;
        public final long expireAt;
        public Ticket(String clientId, String verifyData, long expireAt) {
            this.clientId = clientId;
            this.verifyData = verifyData;
            this.expireAt = expireAt;
        }
    }

    public static Ticket issue(
            String userId, String userName, String nickName,
            String tenantId, String tenantCode, String tenantName,
            String regionCode, String regionName, String remark) {
        long now = System.currentTimeMillis();
        long expireAt = now + TTL_MS;
        Map<String, Object> payload = new LinkedHashMap<String, Object>();
        payload.put("userId", required(userId, "userId"));
        payload.put("userName", required(userName, "userName"));
        payload.put("nickName", required(nickName, "nickName"));
        payload.put("tenantId", required(tenantId, "tenantId"));
        payload.put("tenantCode", required(tenantCode, "tenantCode"));
        payload.put("tenantName", required(tenantName, "tenantName"));
        payload.put("regionCode", required(regionCode, "regionCode"));
        if (regionName != null && !regionName.trim().isEmpty()) {
            payload.put("regionName", regionName.trim());
        }
        payload.put("timestamp", Long.valueOf(now));
        payload.put("expireAt", Long.valueOf(expireAt));
        payload.put("nonce", UUID.randomUUID().toString());
        if (remark != null && !remark.trim().isEmpty()) {
            payload.put("remark", remark.trim());
        }
        byte[] key = Base64.getDecoder().decode(KEY_BASE64);
        String json = toJson(payload);
        String verifyData = encrypt(key, KEY_VERSION, json,
                CLIENT_ID.getBytes(StandardCharsets.UTF_8));
        return new Ticket(CLIENT_ID, verifyData, expireAt);
    }

    public static String encrypt(byte[] keyBytes, int keyVersion, String payload, byte[] aad) {
        try {
            byte[] iv = new byte[IV_LEN];
            RANDOM.nextBytes(iv);
            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(Cipher.ENCRYPT_MODE, new SecretKeySpec(keyBytes, "AES"),
                    new GCMParameterSpec(TAG_BITS, iv));
            if (aad != null) cipher.updateAAD(aad);
            byte[] cipherText = cipher.doFinal(payload.getBytes(StandardCharsets.UTF_8));
            ByteBuffer buf = ByteBuffer.allocate(1 + IV_LEN + cipherText.length);
            buf.put((byte) keyVersion).put(iv).put(cipherText);
            return Base64.getUrlEncoder().withoutPadding().encodeToString(buf.array());
        } catch (Exception e) {
            throw new IllegalStateException("SSO encrypt failed", e);
        }
    }

    public static String jumpUrl(String bOrigin, Ticket ticket) {
        String origin = bOrigin.endsWith("/") ? bOrigin.substring(0, bOrigin.length() - 1) : bOrigin;
        return origin + "/hbcdcagent/workbench/deepseek?clientId=" + ticket.clientId
                + "&verifyData=" + ticket.verifyData;
    }

    static String toJson(Map<String, Object> map) {
        StringBuilder sb = new StringBuilder(256);
        sb.append('{');
        boolean first = true;
        for (Map.Entry<String, Object> e : map.entrySet()) {
            if (!first) sb.append(',');
            first = false;
            sb.append('"').append(escape(e.getKey())).append('"').append(':');
            Object v = e.getValue();
            if (v instanceof Number) sb.append(v.toString());
            else sb.append('"').append(escape(String.valueOf(v))).append('"');
        }
        return sb.append('}').toString();
    }

    private static String required(String value, String field) {
        if (value == null || value.trim().isEmpty()) {
            throw new IllegalArgumentException("payload missing " + field);
        }
        return value.trim();
    }

    private static String escape(String s) {
        StringBuilder sb = new StringBuilder(s.length() + 8);
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            switch (c) {
                case '"':  sb.append("\\\""); break;
                case '\\': sb.append("\\\\"); break;
                case '\n': sb.append("\\n"); break;
                case '\r': sb.append("\\r"); break;
                case '\t': sb.append("\\t"); break;
                default:
                    if (c < 0x20) sb.append(String.format("\\u%04x", Integer.valueOf(c)));
                    else sb.append(c);
            }
        }
        return sb.toString();
    }
}

A 后端调用示例

SsoTicketIssuer.Ticket ticket = SsoTicketIssuer.issue(
        currentUser.getUserId(),
        currentUser.getUserName(),
        currentUser.getNickName(),
        currentUser.getTenantId(),
        currentUser.getTenantCode(),
        currentUser.getTenantName(),
        currentUser.getRegionCode(),
        currentUser.getRegionName(),
        "from-alert");

// 返回给 A 前端
// { "clientId": ticket.clientId, "verifyData": ticket.verifyData, "expireAt": ticket.expireAt }

String url = SsoTicketIssuer.jumpUrl("https://<B域名>", ticket);
